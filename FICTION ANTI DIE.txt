local function saveConfig()
	pcall(function()
		if not writefile then return end
		writefile(CONFIG_FILE, HttpService:JSONEncode({
			active      = active,
			antiFlingOn = antiFlingOn,
			antiDropOn  = antiDropOn,
			uiLocked    = uiLocked,
			isMinimized = isMinimized,
		}))
	end)
end

local function loadConfig()
	pcall(function()
		if not (readfile and isfile and isfile(CONFIG_FILE)) then return end
		local data = HttpService:JSONDecode(readfile(CONFIG_FILE))
		if type(data) ~= "table" then return end
		if data.active == true then active = true end
		if data.antiFlingOn == true then antiFlingOn = true end
		if data.antiDropOn == true then antiDropOn = true end
		if data.uiLocked ~= nil then uiLocked = data.uiLocked end
		if data.isMinimized == true then isMinimized = true end
	end)
end

local _antiDie = {
	loop       = nil,
	healthConn = nil,
	stateConn  = nil,
	diedConn   = nil,
	charConn   = nil,
	invincibleUntil = 0,
	highlight  = nil,
}

local function superHeal(hum)
	if not hum or not hum.Parent then return end
	local maxH = hum.MaxHealth
	if not maxH or maxH <= 0 then maxH = 100 end
	pcall(function() hum.Health = maxH end)
	_antiDie.invincibleUntil = tick() + 0.5
	pcall(function()
		local char = hum.Parent
		if not char then return end
		for _, c in ipairs(char:GetChildren()) do
			if c:IsA("NumberValue") or c:IsA("IntValue") then
				local n = c.Name:lower()
				if n:find("health") or n:find("hp") or n:find("life") then
					c.Value = maxH
				end
			elseif c:IsA("BoolValue") then
				local n = c.Name:lower()
				if n:find("dead") or n:find("died") then
					c.Value = false
				end
			end
		end
		pcall(function()
			if char:GetAttribute("Dead") ~= nil then char:SetAttribute("Dead", false) end
			if char:GetAttribute("IsDead") ~= nil then char:SetAttribute("IsDead", false) end
			if char:GetAttribute("Health") ~= nil then char:SetAttribute("Health", maxH) end
		end)
	end)
end

local function forceAlive(hum, root)
	if not hum then return end
	superHeal(hum)
	pcall(function()
		hum:ChangeState(Enum.HumanoidStateType.Running)
		hum.PlatformStand = false
		hum.Sit = false
		hum.AutoRotate = true
		hum.BreakJointsOnDeath = false
	end)
	if root and root.Parent then
		pcall(function()
			root.AssemblyAngularVelocity = Vector3.zero
			root.AssemblyLinearVelocity = Vector3.new(0, math.max(root.AssemblyLinearVelocity.Y, 2), 0)
			root.CFrame = CFrame.new(root.Position + Vector3.new(0, 2.5, 0))
		end)
	end
end

local function preventDamage(root, hum)
	if not hum or not hum.Parent then return end
	local maxH = hum.MaxHealth or 100
	if tick() < (_antiDie.invincibleUntil or 0) then
		if hum.Health < maxH then hum.Health = maxH end
	end
	local st = hum:GetState()
	if st == Enum.HumanoidStateType.Physics
		or st == Enum.HumanoidStateType.Ragdoll
		or st == Enum.HumanoidStateType.FallingDown
		or st == Enum.HumanoidStateType.Dead then
		forceAlive(hum, root)
	end
	if hum.Health <= 0 then
		forceAlive(hum, root)
	end
end

local function applyHugeHealth(hum)
	if not hum or not hum.Parent then return end
	pcall(function()
		hum.MaxHealth = math.huge
		hum.Health = math.huge
		hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
		hum.BreakJointsOnDeath = false
	end)
end

local function opRecovery(hum, char)
	if not hum or not hum.Parent then return end
	if hum.Health <= 0 then
		pcall(function()
			hum.MaxHealth = math.huge
			hum.Health = math.huge
			hum.BreakJointsOnDeath = false
		end)
	end
	if hum.Health < 10 then
		pcall(function()
			hum.PlatformStand = false
			hum.Sit = false
			hum.AutoRotate = true
		end)
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if hrp then
			pcall(function()
				hrp.AssemblyLinearVelocity = Vector3.new(
					hrp.AssemblyLinearVelocity.X,
					math.max(hrp.AssemblyLinearVelocity.Y, 0),
					hrp.AssemblyLinearVelocity.Z
				)
			end)
		end
	end
	if hum.Health <= 0 and hum:GetState() == Enum.HumanoidStateType.Dead then
		pcall(function()
			hum:ChangeState(Enum.HumanoidStateType.Running)
			hum.Health = math.huge
		end)
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if hrp then
			pcall(function()
				hrp.CFrame = CFrame.new(hrp.Position + Vector3.new(0, 8, 0))
				hrp.AssemblyLinearVelocity = Vector3.zero
				hrp.AssemblyAngularVelocity = Vector3.zero
			end)
		end
	end
end

local function ensureHighlight(char)
	if not char then return end
	if _antiDie.highlight and _antiDie.highlight.Parent then return end
	pcall(function()
		if _antiDie.highlight then _antiDie.highlight:Destroy() end
		local hl = Instance.new("Highlight")
		hl.Name = "FictionAllAntiDieHighlight"
		hl.FillColor = Color3.fromRGB(255, 255, 255)
		hl.FillTransparency = 0.7
		hl.OutlineColor = Color3.fromRGB(255, 255, 255)
		hl.OutlineTransparency = 0.2
		hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		hl.Parent = char
		_antiDie.highlight = hl
	end)
end

local function clearHighlight()
	if _antiDie.highlight then
		pcall(function() _antiDie.highlight:Destroy() end)
		_antiDie.highlight = nil
	end
end

local function disconnectAll()
	local A = _antiDie
	if A.loop then pcall(function() A.loop:Disconnect() end); A.loop = nil end
	if A.healthConn then pcall(function() A.healthConn:Disconnect() end); A.healthConn = nil end
	if A.stateConn then pcall(function() A.stateConn:Disconnect() end); A.stateConn = nil end
	if A.diedConn then pcall(function() A.diedConn:Disconnect() end); A.diedConn = nil end
end

local function monitorChar(char)
	if not char then return end
	local A = _antiDie
	if A.healthConn then pcall(function() A.healthConn:Disconnect() end); A.healthConn = nil end
	if A.stateConn then pcall(function() A.stateConn:Disconnect() end); A.stateConn = nil end
	if A.diedConn then pcall(function() A.diedConn:Disconnect() end); A.diedConn = nil end

	local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 3)
	if not hum then return end

	applyHugeHealth(hum)
	ensureHighlight(char)

	A.healthConn = hum:GetPropertyChangedSignal("Health"):Connect(function()
		if not active then return end
		applyHugeHealth(hum)
		if hum.Health <= 0 then
			forceAlive(hum, char:FindFirstChild("HumanoidRootPart"))
			opRecovery(hum, char)
		elseif hum.Health <= 25 then
			superHeal(hum)
			applyHugeHealth(hum)
		end
	end)

	A.stateConn = hum.StateChanged:Connect(function(_, new)
		if not active then return end
		if new == Enum.HumanoidStateType.Dead
			or new == Enum.HumanoidStateType.Physics
			or new == Enum.HumanoidStateType.Ragdoll
			or new == Enum.HumanoidStateType.FallingDown then
			applyHugeHealth(hum)
			forceAlive(hum, char:FindFirstChild("HumanoidRootPart"))
			opRecovery(hum, char)
		end
	end)

	A.diedConn = hum.Died:Connect(function()
		if not active then return end
		task.defer(function()
			applyHugeHealth(hum)
			forceAlive(hum, char:FindFirstChild("HumanoidRootPart"))
			opRecovery(hum, char)
		end)
	end)

	superHeal(hum)
	applyHugeHealth(hum)
end

local function startAntiDie()
	active = true
	disconnectAll()
	local A = _antiDie

	A.loop = RunService.Heartbeat:Connect(function()
		if not active then return end
		local char = LP.Character
		if not char then return end
		local hum = char:FindFirstChildOfClass("Humanoid")
		local root = char:FindFirstChild("HumanoidRootPart")
		if not hum then return end

		applyHugeHealth(hum)

		if hum.Health <= 0 then
			forceAlive(hum, root)
			opRecovery(hum, char)
			return
		end
		if hum.Health <= 25 then
			superHeal(hum)
			applyHugeHealth(hum)
		end
		preventDamage(root, hum)
		if hum.Health < 20 and hum.Health > 0 then
			superHeal(hum)
			applyHugeHealth(hum)
		end
		opRecovery(hum, char)
		if not A.highlight or not A.highlight.Parent then
			ensureHighlight(char)
		end
	end)

	if LP.Character then monitorChar(LP.Character) end
	if not A.charConn then
		A.charConn = LP.CharacterAdded:Connect(function(char)
			if not active then return end
			task.wait(0.08)
			monitorChar(char)
		end)
	end
end

local function stopAntiDie()
	active = false
	disconnectAll()
	clearHighlight()
	local char = LP.Character
	if char then
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then
			pcall(function()
				hum:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
				hum.MaxHealth = 100
				hum.Health = 100
				hum.BreakJointsOnDeath = true
			end)
		end
	end
end

local AntiFlingPlayersConns = {}
local AntiFlingHeartbeat = nil
local AntiFlingPlayerAdded = nil
local AntiFlingPlayerRemoving = nil

local function LimitPlayerVelocity(part)
	local lv = part.AssemblyLinearVelocity
	local av = part.AssemblyAngularVelocity
	if lv.Magnitude > 100 then part.AssemblyLinearVelocity = lv.Unit * 100 end
	if av.Magnitude > 75 then part.AssemblyAngularVelocity = av.Unit * 75 end
end

local function SetupPlayerAntiFling(player)
	if AntiFlingPlayersConns[player] then pcall(function() AntiFlingPlayersConns[player]:Disconnect() end) end
	AntiFlingPlayersConns[player] = player.CharacterAdded:Connect(function(char)
		if antiFlingOn then
			task.wait(0.25)
			local hrp = char:WaitForChild("HumanoidRootPart", 10)
			if hrp then pcall(function() hrp:SetNetworkOwner(nil) end) end
		end
	end)
	if player.Character then
		local hrp = player.Character:FindFirstChild("HumanoidRootPart")
		if hrp then pcall(function() hrp:SetNetworkOwner(nil) end) end
	end
end

local function startAntiFling()
	antiFlingOn = true
	for _, p in ipairs(Players:GetPlayers()) do SetupPlayerAntiFling(p) end
	AntiFlingPlayerAdded = Players.PlayerAdded:Connect(function(p)
		if antiFlingOn then SetupPlayerAntiFling(p) end
	end)
	AntiFlingPlayerRemoving = Players.PlayerRemoving:Connect(function(p)
		if AntiFlingPlayersConns[p] then
			pcall(function() AntiFlingPlayersConns[p]:Disconnect() end)
			AntiFlingPlayersConns[p] = nil
		end
	end)
	AntiFlingHeartbeat = RunService.Heartbeat:Connect(function()
		if not antiFlingOn then return end
		for _, p in ipairs(Players:GetPlayers()) do
			if p.Character then
				local hrp = p.Character:FindFirstChild("HumanoidRootPart")
				if hrp then LimitPlayerVelocity(hrp) end
			end
		end
	end)
end

local function stopAntiFling()
	antiFlingOn = false
	if AntiFlingHeartbeat then AntiFlingHeartbeat:Disconnect(); AntiFlingHeartbeat = nil end
	if AntiFlingPlayerAdded then AntiFlingPlayerAdded:Disconnect(); AntiFlingPlayerAdded = nil end
	if AntiFlingPlayerRemoving then AntiFlingPlayerRemoving:Disconnect(); AntiFlingPlayerRemoving = nil end
	for _, c in pairs(AntiFlingPlayersConns) do pcall(function() c:Disconnect() end) end
	AntiFlingPlayersConns = {}
end

local RawMetatableHooked = false
local OldIndex = nil
local OldNewIndex = nil
local DroppedVelocity = Vector3.zero

local function startAntiDrop()
	antiDropOn = true
	if RawMetatableHooked then return end
	if not getrawmetatable then return end
	local mt = getrawmetatable(game)
	if not mt then return end
	OldIndex = mt.__index
	OldNewIndex = mt.__newindex
	setreadonly(mt, false)
	mt.__index = newcclosure(function(t, k)
		if not checkcaller() and (k == "AssemblyLinearVelocity" or k == "Velocity") and typeof(t) == "Instance" and t:IsA("BasePart") and t.Name == "HumanoidRootPart" and LP.Character and t:IsDescendantOf(LP.Character) then
			return DroppedVelocity
		end
		return OldIndex(t, k)
	end)
	mt.__newindex = newcclosure(function(t, k, v)
		if not checkcaller() and (k == "AssemblyLinearVelocity" or k == "Velocity") and typeof(t) == "Instance" and t:IsA("BasePart") and t.Name == "HumanoidRootPart" and LP.Character and t:IsDescendantOf(LP.Character) then
			DroppedVelocity = v
			return
		end
		return OldNewIndex(t, k, v)
	end)
	setreadonly(mt, true)
	RawMetatableHooked = true
end

local function stopAntiDrop()
	antiDropOn = false
	if not RawMetatableHooked then return end
	local mt = getrawmetatable(game)
	if mt and OldIndex then
		setreadonly(mt, false)
		mt.__index = OldIndex
		mt.__newindex = OldNewIndex
		setreadonly(mt, true)
		OldIndex = nil
		OldNewIndex = nil
	end
	RawMetatableHooked = false
end

	stroke.Color = Color3.fromRGB(255, 255, 255)

	grad.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0.00, Color3.fromRGB(255, 255, 255)),
		ColorSequenceKeypoint.new(0.25, Color3.fromRGB(80, 80, 80)),
		ColorSequenceKeypoint.new(0.50, Color3.fromRGB(255, 255, 255)),
		ColorSequenceKeypoint.new(0.75, Color3.fromRGB(80, 80, 80)),
		ColorSequenceKeypoint.new(1.00, Color3.fromRGB(255, 255, 255)),
	})
	grad.Rotation = 0
	grad.Parent = stroke
	table.insert(rotatingStrokes, grad)
	return stroke
end

end)

local pingFillGrads = {}
local pingStrokeGrads = {}

local function pingOn(inst)
	if not inst or not inst.Parent then return end
	if inst:GetAttribute("_PingOn") then return end
	inst:SetAttribute("_PingOn", true)
	pcall(function()
		inst.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
		inst.BackgroundTransparency = 0
	end)
	local fill = Instance.new("UIGradient")
	fill.Name = "_PingFill"
	fill.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0.00, Color3.fromRGB(200, 200, 200)),
		ColorSequenceKeypoint.new(0.20, Color3.fromRGB(255, 255, 255)),
		ColorSequenceKeypoint.new(0.40, Color3.fromRGB(180, 180, 180)),
		ColorSequenceKeypoint.new(0.60, Color3.fromRGB(255, 255, 255)),
		ColorSequenceKeypoint.new(0.80, Color3.fromRGB(200, 200, 200)),
		ColorSequenceKeypoint.new(1.00, Color3.fromRGB(255, 255, 255)),
	})
	fill.Rotation = 0
	fill.Parent = inst
	table.insert(pingFillGrads, fill)
	local stroke = inst:FindFirstChildWhichIsA("UIStroke")
	if stroke then
		local sg = Instance.new("UIGradient")
		sg.Name = "_PingStroke"
		sg.Color = ColorSequence.new({
			ColorSequenceKeypoint.new(0.00, Color3.fromRGB(255, 255, 255)),
			ColorSequenceKeypoint.new(0.35, Color3.fromRGB(120, 120, 120)),
			ColorSequenceKeypoint.new(0.65, Color3.fromRGB(255, 255, 255)),
			ColorSequenceKeypoint.new(1.00, Color3.fromRGB(80, 80, 80)),
		})
		sg.Parent = stroke
		table.insert(pingStrokeGrads, sg)
	end
end

local function pingOff(inst)
	if not inst or not inst.Parent then return end
	if not inst:GetAttribute("_PingOn") then return end
	inst:SetAttribute("_PingOn", nil)
	local fill = inst:FindFirstChild("_PingFill")
	if fill then fill:Destroy() end
	local stroke = inst:FindFirstChildWhichIsA("UIStroke")
	if stroke then
		local ss = stroke:FindFirstChild("_PingStroke")
		if ss then ss:Destroy() end
	end
end

		end
	end
end)

UserInputService.InputBegan:Connect(function(input, processed)
	if processed then return end
	if input.KeyCode == Enum.KeyCode.LeftControl then
		main.Visible = not main.Visible
	end
end)

do
	local dragging, dragStart, startPos
	header.InputBegan:Connect(function(input)
		if uiLocked then return end
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = main.Position
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			local delta = input.Position - dragStart
			main.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y
			)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end
